avg <- imp_data %>%
  group_by(`Terrain Attribute`) %>%
  summarise(total = mean(MSE_rank)) %>%
  arrange(total) |>
  mutate(MSE_rank = 1:n(),
         property = "Overall")

avg_site<- imp_data %>%
  group_by(`Terrain Attribute`, site) %>%
  summarise(total = mean(MSE_rank)) %>%
  group_by(site) %>%
  arrange(total) |>
  mutate(MSE_rank = 1:n(),
         property = site)
